cmake_minimum_required(VERSION 3.21)
project(LuminPackNative LANGUAGES CXX)
set(LUMINPACK_HEADER "" CACHE FILEPATH "Generated C++ root header")
set(LUMINPACK_MANIFEST "" CACHE FILEPATH "Exported managed manifest")
set(LUMINPACK_ROOT "" CACHE STRING "Managed root identity")
option(LUMINPACK_RTTI "Generate RTTI Union dispatch" OFF)
set(LUMINPACK_REGISTRATION "" CACHE FILEPATH "Header defining luminpack_register_model() after the generated root")
set(LUMINPACK_BINDINGS "" CACHE FILEPATH "Optional application bindings included before the root")
if(LUMINPACK_HEADER)
    get_filename_component(header "${LUMINPACK_HEADER}" ABSOLUTE)
elseif(LUMINPACK_MANIFEST AND LUMINPACK_ROOT)
    set(header "${CMAKE_CURRENT_BINARY_DIR}/root.hpp")
    get_filename_component(manifest "${LUMINPACK_MANIFEST}" ABSOLUTE)
    if(LUMINPACK_RTTI)
        set(options --rtti)
    endif()
    execute_process(COMMAND dotnet run -c Release --project "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/LuminPack.Codegen"
        -- generate --language cpp ${options} "${manifest}" "${LUMINPACK_ROOT}" "${header}" COMMAND_ERROR_IS_FATAL ANY)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${manifest}")
else()
    message(FATAL_ERROR "Set LUMINPACK_MANIFEST and LUMINPACK_ROOT or supply LUMINPACK_HEADER")
endif()
file(STRINGS "${header}" namespace REGEX "^namespace luminpack_generated::root_[A-Za-z0-9_]+$")
if(NOT namespace)
    message(FATAL_ERROR "Expected a generated LuminPack C++ root header")
endif()
string(REPLACE "namespace " "" namespace "${namespace}")
file(STRINGS "${header}" graph REGEX "root_requires_reference_graph = true")
file(STRINGS "${header}" facade REGEX "inline auto read_root_binary")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(../LuminPack.Cpp cpp)
add_library(luminpack_model SHARED model.cpp)
target_link_libraries(luminpack_model PRIVATE LuminPack::Cpp)
target_compile_definitions(luminpack_model PRIVATE LP_ROOT=${namespace})
target_include_directories(luminpack_model PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${header}")
if(graph)
    target_compile_definitions(luminpack_model PRIVATE LP_GRAPH)
endif()
if(facade)
    target_compile_definitions(luminpack_model PRIVATE LP_FACADE)
endif()
if(LUMINPACK_BINDINGS)
    get_filename_component(bindings "${LUMINPACK_BINDINGS}" ABSOLUTE)
    set(binding_include "#include \"${bindings}\"")
endif()
if(LUMINPACK_REGISTRATION)
    get_filename_component(registration "${LUMINPACK_REGISTRATION}" ABSOLUTE)
    set(registration_include "#include \"${registration}\"")
    target_compile_definitions(luminpack_model PRIVATE LP_REGISTER)
endif()
configure_file(model_header.h.in model_header.h @ONLY)
