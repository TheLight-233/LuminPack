cmake_minimum_required(VERSION 3.21)
project(LuminPackCpp LANGUAGES CXX)

option(LUMINPACK_CPP_SIMD "Enable CPU-dispatched text and layout acceleration" ON)
option(LUMINPACK_CPP_TESTS "Build the C++ wire tests" ${PROJECT_IS_TOP_LEVEL})
find_package(Threads REQUIRED)
set(LUMINPACK_CPP_MAX_SIMD_BITS 512 CACHE STRING "Maximum compiled x64 SIMD width")
set_property(CACHE LUMINPACK_CPP_MAX_SIMD_BITS PROPERTY STRINGS 128 256 512)
if(NOT LUMINPACK_CPP_MAX_SIMD_BITS MATCHES "^(128|256|512)$")
    message(FATAL_ERROR "LUMINPACK_CPP_MAX_SIMD_BITS must be 128, 256, or 512.")
endif()

add_library(luminpack_cpp STATIC
    src/luminpack.cpp src/latin1.cpp src/layout_conversion.cpp src/string_profiles.cpp src/utf8_runs.cpp src/version.cpp
    src/json.cpp src/json_scan.cpp src/xxhash3.cpp src/compression.cpp src/compression_match.cpp)
add_library(LuminPack::Cpp ALIAS luminpack_cpp)
target_compile_features(luminpack_cpp PUBLIC cxx_std_20)
target_include_directories(luminpack_cpp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(luminpack_cpp PUBLIC Threads::Threads)

# Keep the baseline translation units free of AVX instructions.
if(LUMINPACK_CPP_SIMD)
    target_compile_definitions(luminpack_cpp PRIVATE LUMINPACK_CPP_SIMD)
    set(luminpack_arch "${CMAKE_SYSTEM_PROCESSOR}")
    if(CMAKE_VS_PLATFORM_NAME)
        set(luminpack_arch "${CMAKE_VS_PLATFORM_NAME}")
    elseif(CMAKE_OSX_ARCHITECTURES)
        set(luminpack_arch "${CMAKE_OSX_ARCHITECTURES}")
    elseif(CMAKE_CXX_COMPILER_TARGET)
        set(luminpack_arch "${CMAKE_CXX_COMPILER_TARGET}")
    endif()
    if(CMAKE_SIZEOF_VOID_P EQUAL 8 AND luminpack_arch MATCHES "^(AMD64|amd64|x64|x86_64)(-|$)")
        target_compile_definitions(luminpack_cpp PUBLIC LUMINPACK_CPP_INLINE_SSE2)
        if(MSVC OR CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            set(luminpack_ssse3_sources src/layout_conversion_ssse3.cpp src/utf8_runs_ssse3.cpp)
            target_sources(luminpack_cpp PRIVATE ${luminpack_ssse3_sources})
            target_compile_definitions(luminpack_cpp PRIVATE LUMINPACK_CPP_SSSE3)
            if(MSVC AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
                set_source_files_properties(${luminpack_ssse3_sources} PROPERTIES COMPILE_OPTIONS "/clang:-mssse3")
            elseif(NOT MSVC)
                set_source_files_properties(${luminpack_ssse3_sources} PROPERTIES COMPILE_OPTIONS "-mssse3")
            endif()
        endif()
        if(MSVC)
            set(luminpack_avx2_flag /arch:AVX2)
            set(luminpack_avx512f_flag /arch:AVX512)
            set(luminpack_avx512_flags /arch:AVX512)
        elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            set(luminpack_avx2_flag -mavx2)
            set(luminpack_avx512f_flag -mavx512f)
            set(luminpack_avx512_flags -mavx512f -mavx512bw)
        endif()
        if(luminpack_avx2_flag AND LUMINPACK_CPP_MAX_SIMD_BITS GREATER_EQUAL 256)
            set(luminpack_avx2_sources
                src/latin1_avx2.cpp src/json_scan_avx2.cpp src/layout_conversion_avx2.cpp
                src/utf8_ascii_read_avx2.cpp src/utf8_runs_avx2.cpp src/xxhash3_avx2.cpp src/compression_match_avx2.cpp)
            target_sources(luminpack_cpp PRIVATE ${luminpack_avx2_sources})
            set_source_files_properties(${luminpack_avx2_sources} PROPERTIES COMPILE_OPTIONS "${luminpack_avx2_flag}")
            target_compile_definitions(luminpack_cpp PRIVATE LUMINPACK_CPP_AVX2)
        endif()
        if(luminpack_avx512_flags AND LUMINPACK_CPP_MAX_SIMD_BITS EQUAL 512)
            set(luminpack_avx512_sources src/latin1_avx512.cpp src/json_scan_avx512.cpp
                src/layout_conversion_avx512.cpp src/layout_conversion_avx512vbmi.cpp src/utf8_runs_avx512.cpp
                src/compression_match_avx512.cpp)
            target_sources(luminpack_cpp PRIVATE ${luminpack_avx512_sources} src/layout_conversion_avx512f.cpp)
            set_source_files_properties(${luminpack_avx512_sources} PROPERTIES COMPILE_OPTIONS "${luminpack_avx512_flags}")
            set_source_files_properties(src/layout_conversion_avx512f.cpp PROPERTIES COMPILE_OPTIONS "${luminpack_avx512f_flag}")
            if(MSVC AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
                set_property(SOURCE src/layout_conversion_avx512vbmi.cpp APPEND PROPERTY COMPILE_OPTIONS "/clang:-mavx512vbmi")
                set_property(SOURCE src/layout_conversion_avx512f.cpp APPEND PROPERTY COMPILE_OPTIONS "/clang:-mno-avx512bw")
            elseif(NOT MSVC)
                set_property(SOURCE src/layout_conversion_avx512vbmi.cpp APPEND PROPERTY COMPILE_OPTIONS "-mavx512vbmi")
            endif()
            target_compile_definitions(luminpack_cpp PRIVATE LUMINPACK_CPP_AVX512BW LUMINPACK_CPP_LAYOUT_AVX512)
        endif()
    endif()
endif()

if(LUMINPACK_CPP_TESTS)
    enable_testing()
    add_subdirectory(../../tests/LuminPack.Tests.Cpp "${CMAKE_CURRENT_BINARY_DIR}/tests")
endif()
