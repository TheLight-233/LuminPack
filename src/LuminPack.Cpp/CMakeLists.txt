cmake_minimum_required(VERSION 3.21)
project(LuminPackCpp LANGUAGES CXX)

option(LUMINPACK_CPP_SIMD "Enable CPU-dispatched Latin-1 acceleration" ON)
option(LUMINPACK_CPP_TESTS "Build the C++ wire tests" ${PROJECT_IS_TOP_LEVEL})
set(LUMINPACK_CPP_MAX_SIMD_BITS 512 CACHE STRING "Maximum compiled x64 SIMD width")
set_property(CACHE LUMINPACK_CPP_MAX_SIMD_BITS PROPERTY STRINGS 128 256 512)
if(NOT LUMINPACK_CPP_MAX_SIMD_BITS MATCHES "^(128|256|512)$")
    message(FATAL_ERROR "LUMINPACK_CPP_MAX_SIMD_BITS must be 128, 256, or 512.")
endif()

add_library(luminpack_cpp STATIC src/luminpack.cpp src/latin1.cpp src/string_profiles.cpp src/version.cpp src/json.cpp)
add_library(LuminPack::Cpp ALIAS luminpack_cpp)
target_compile_features(luminpack_cpp PUBLIC cxx_std_20)
target_include_directories(luminpack_cpp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")

# Keep the baseline translation units free of AVX instructions.
if(LUMINPACK_CPP_SIMD)
    target_compile_definitions(luminpack_cpp PRIVATE LUMINPACK_CPP_SIMD)
    set(luminpack_arch "${CMAKE_SYSTEM_PROCESSOR}")
    if(CMAKE_VS_PLATFORM_NAME)
        set(luminpack_arch "${CMAKE_VS_PLATFORM_NAME}")
    elseif(CMAKE_OSX_ARCHITECTURES)
        set(luminpack_arch "${CMAKE_OSX_ARCHITECTURES}")
    elseif(CMAKE_CXX_COMPILER_TARGET)
        set(luminpack_arch "${CMAKE_CXX_COMPILER_TARGET}")
    endif()
    if(CMAKE_SIZEOF_VOID_P EQUAL 8 AND luminpack_arch MATCHES "^(AMD64|amd64|x64|x86_64)(-|$)")
        if(MSVC)
            set(luminpack_avx2_flag /arch:AVX2)
            set(luminpack_avx512_flags /arch:AVX512)
        elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            set(luminpack_avx2_flag -mavx2)
            set(luminpack_avx512_flags -mavx512f -mavx512bw)
        endif()
        if(luminpack_avx2_flag AND LUMINPACK_CPP_MAX_SIMD_BITS GREATER_EQUAL 256)
            target_sources(luminpack_cpp PRIVATE src/latin1_avx2.cpp)
            set_source_files_properties(src/latin1_avx2.cpp PROPERTIES COMPILE_OPTIONS "${luminpack_avx2_flag}")
            target_compile_definitions(luminpack_cpp PRIVATE LUMINPACK_CPP_AVX2)
        endif()
        if(luminpack_avx512_flags AND LUMINPACK_CPP_MAX_SIMD_BITS EQUAL 512)
            target_sources(luminpack_cpp PRIVATE src/latin1_avx512.cpp)
            set_source_files_properties(src/latin1_avx512.cpp PROPERTIES COMPILE_OPTIONS "${luminpack_avx512_flags}")
            target_compile_definitions(luminpack_cpp PRIVATE LUMINPACK_CPP_AVX512BW)
        endif()
    endif()
endif()

if(LUMINPACK_CPP_TESTS)
    enable_testing()
    add_subdirectory(../../tests/LuminPack.Tests.Cpp "${CMAKE_CURRENT_BINARY_DIR}/tests")
endif()
