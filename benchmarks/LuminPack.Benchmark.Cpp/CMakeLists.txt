cmake_minimum_required(VERSION 3.21)
project(LuminPackCppBenchmark LANGUAGES CXX)

set(LUMINPACK_CPP_MODEL_FIXTURES "" CACHE PATH "Generated benchmark headers and .NET fixtures")
set(LUMINPACK_CPP_YALANTINGLIBS "" CACHE PATH "Checkout of yalantinglibs 0.5.8")
set(LUMINPACK_CPP_RAPIDJSON "" CACHE PATH "Checkout of RapidJSON (tested with 297637b)")
set(LUMINPACK_CPP_BITSERY "" CACHE PATH "Optional checkout of Bitsery 5.2.5")
set(LUMINPACK_CPP_CEREAL "" CACHE PATH "Optional checkout of cereal 1.3.2")
set(LUMINPACK_CPP_MSGPACK "" CACHE PATH "Optional checkout of MessagePack-C++ 7.0.0")
set(LUMINPACK_CPP_JSONCONS "" CACHE PATH "Optional checkout of jsoncons 1.3.2")
set(LUMINPACK_CPP_NLOHMANN_JSON "" CACHE PATH "Optional checkout of nlohmann/json 3.12.0")
set(LUMINPACK_CPP_SIMDJSON "" CACHE PATH "Optional checkout of simdjson 3.12.3")
foreach(file IN ITEMS
    "${LUMINPACK_CPP_MODEL_FIXTURES}/simple-benchmark.hpp"
    "${LUMINPACK_CPP_MODEL_FIXTURES}/serialize-benchmark.hpp"
    "${LUMINPACK_CPP_YALANTINGLIBS}/include/ylt/struct_pack.hpp"
    "${LUMINPACK_CPP_RAPIDJSON}/include/rapidjson/document.h")
    if(NOT EXISTS "${file}")
        message(FATAL_ERROR "Missing benchmark dependency: ${file}")
    endif()
endforeach()

add_subdirectory(../../src/LuminPack.Cpp "${CMAKE_CURRENT_BINARY_DIR}/luminpack")
add_executable(luminpack_cpp_benchmark benchmark.cpp allocations.cpp)
target_link_libraries(luminpack_cpp_benchmark PRIVATE LuminPack::Cpp)
target_include_directories(luminpack_cpp_benchmark PRIVATE "${LUMINPACK_CPP_MODEL_FIXTURES}")
target_include_directories(luminpack_cpp_benchmark SYSTEM PRIVATE
    "${LUMINPACK_CPP_YALANTINGLIBS}/include"
    "${LUMINPACK_CPP_YALANTINGLIBS}/include/ylt/thirdparty"
    "${LUMINPACK_CPP_RAPIDJSON}/include")

function(add_benchmark_reference name header)
    set(checkout "${LUMINPACK_CPP_${name}}")
    if(checkout)
        if(NOT EXISTS "${checkout}/include/${header}")
            message(FATAL_ERROR "Missing benchmark dependency: ${checkout}/include/${header}")
        endif()
        target_include_directories(luminpack_cpp_benchmark SYSTEM PRIVATE "${checkout}/include")
        target_compile_definitions(luminpack_cpp_benchmark PRIVATE "LUMINPACK_BENCHMARK_${name}")
    endif()
endfunction()

add_benchmark_reference(BITSERY bitsery/bitsery.h)
add_benchmark_reference(CEREAL cereal/archives/binary.hpp)
add_benchmark_reference(MSGPACK msgpack.hpp)
add_benchmark_reference(JSONCONS jsoncons/json.hpp)
add_benchmark_reference(NLOHMANN_JSON nlohmann/json.hpp)
if(LUMINPACK_CPP_MSGPACK)
    target_compile_definitions(luminpack_cpp_benchmark PRIVATE MSGPACK_NO_BOOST MSGPACK_USE_STD_VARIANT_ADAPTOR)
endif()
if(LUMINPACK_CPP_SIMDJSON)
    set(simdjson_sources "${LUMINPACK_CPP_SIMDJSON}/singleheader")
    foreach(file IN ITEMS simdjson.h simdjson.cpp)
        if(NOT EXISTS "${simdjson_sources}/${file}")
            message(FATAL_ERROR "Missing benchmark dependency: ${simdjson_sources}/${file}")
        endif()
    endforeach()
    target_include_directories(luminpack_cpp_benchmark SYSTEM PRIVATE "${simdjson_sources}")
    target_sources(luminpack_cpp_benchmark PRIVATE "${simdjson_sources}/simdjson.cpp")
    target_compile_definitions(luminpack_cpp_benchmark PRIVATE LUMINPACK_BENCHMARK_SIMDJSON)
endif()
if(MSVC)
    target_compile_options(luminpack_cpp_benchmark PRIVATE /bigobj /Zc:__cplusplus)
endif()
